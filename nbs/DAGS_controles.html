Vamos a mostrar que el enfoque cuantitativo "puro" falla. Comenzamos con una regresión con controles en los cuatro escenarios simulados. Medicion de efectos observados:

```{webr-r}
#| message: false
#| warning: false
library(quartets)
library(tidyverse)
library(broom)
library(tidyverse)

# 1. Tu resumen estadístico (ampliado con Sesgo)
analisis_causal <- causal_quartet |> 
  group_by(dataset) |> 
  summarise(
    beta_simple = coef(lm(outcome ~ exposure))[2],
    beta_ajustada = coef(lm(outcome ~ exposure + covariate))[2],
    cor_xz = cor(exposure, covariate)
  ) |> 
  mutate(
    # En este cuarteto, el efecto verdadero suele ser 1
    sesgo_ajuste = abs(1 - beta_ajustada)
  )

print(analisis_causal)
```

Simulamos un efecto total de `x` sobre `y` igual a 6, pero totalmente mediado sólo por `z`, y que el efecto directo de `x` sobre `y` es 0.

```{webr-r}

x <- rnorm(800)
z <-  3*x + rnorm(800)
y <- 2*z + rnorm(800)
d <- tibble(x, y, z)

# Para valores similares de Z el efecto de X sobre Y es 0, 
# Se puede observar haciendo  "cortes" en Z para verlo
d  |>
  mutate(z_cut = cut_interval(z, 9)) |>
 ggplot(aes(x = x, y = y)) +
  geom_point(aes(color = z_cut)) +
  geom_smooth(method = "lm", se = FALSE, color = "black", linewidth= 2) +
  geom_smooth(aes(color= z_cut), method = "lm", se = FALSE)

```

Efecto total es sin ajustar por `z` y el efecto directo es ajustando por `z`

```{webr-r}

# efecto total
lm(y ~ x, data = d) |> confint()

# efecto directo
lm(y ~ x + z, data = d) |> confint()

```

```{webr-r}
f <- function(bzx = 2, bxy =2, n = 100){
  z <- rnorm(n, 4, 2)
  x <-  2 + bzx*z + rnorm(n)
  y <-  2 + bxy*x + rnorm(n)
  d <- tibble(x, y, z)
  bx_preciso = coef(lm(y ~ x, data = d))[2]
  bx_parasito = coef(lm(y ~ x + z,  data = d))[2]
 return(c(
    unlist(bx_preciso), unlist(bx_parasito)))
}


sim <- replicate(400, f(n = 200))

d <- data.frame(bx_preciso = sim[1,], bx_parasito = sim[2,])

d <- d  |> 
  pivot_longer(cols =everything())

d  |> 
  ggplot(aes(x = value, fill = name)) +
  geom_density()





```

```{webr-r}
 return(c(
```
